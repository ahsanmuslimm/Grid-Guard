6.3 Wire NVD CVE API Required

NVD (National Vulnerability Database) is free. Get API key at https://nvd.nist.gov/developers/request-an-api-key for higher rate limits.

tools/cve_lookup.py — skeleton
import requests, os NVD_BASE_URL = "https://services.nvd.nist.gov/rest/json/cves/2.0" def lookup_cve(attack_type: str, affected_system: str = "SCADA") -> dict: """Search NVD for CVEs matching the attack type""" params = { "keywordSearch": f"{attack_type} {affected_system} ICS", "resultsPerPage": 5, "cvssV3Severity": "HIGH" } headers = {} if os.getenv("NVD_API_KEY"): headers["apiKey"] = os.getenv("NVD_API_KEY") response = requests.get(NVD_BASE_URL, params=params, headers=headers) data = response.json() cves = [v["cve"]["id"] for v in data.get("vulnerabilities", [])] return {"cves": cves, "total_found": data.get("totalResults", 0)}
